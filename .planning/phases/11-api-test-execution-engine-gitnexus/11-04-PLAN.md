    ```python
    class OpenAPIParser:
        """Parses OpenAPI specs and auto-creates folder structures.

        For each tag in the spec:
        1. Creates a FolderType.API_TEST folder under the project root
        2. For each endpoint in the tag, creates a sub-folder (optional) or APIEndpoint record
        3. Returns summary of created resources
        """

        async def parse_and_create_structure(
            self,
            project_id: UUID,
            spec_url: str,
            parent_folder_id: UUID | None = None,
        ) -> dict:
            """Full pipeline: fetch spec -> parse -> create folders + endpoints.

            Returns:
                {
                    "title": "...",
                    "version": "...",
                    "base_url": "...",
                    "tags_created": 5,
                    "endpoints_created": 23,
                    "folders": [{"id": "...", "name": "...", "tag": "..."}],
                }
            """
            # 1. Use existing parse_api_spec to fetch and parse
            parsed = await parse_api_spec(spec_url)

            # 2. Group operations by tag
            tag_groups = {}  # tag -> [operations]
            for op in parsed.get("operations", []):
                for tag in op.get("tags", ["Untagged"]):
                    tag_groups.setdefault(tag, []).append(op)

            # 3. Create folders and endpoints using async_session_factory
            async with async_session_factory() as session:
                folders_created = []
                endpoints_created = 0

                for tag, operations in tag_groups.items():
                    # Create tag-level folder
                    folder = Folder(
                        project_id=project_id,
                        parent_id=parent_folder_id,
                        name=tag,
                        folder_type=FolderType.API_TEST,
                    )
                    session.add(folder)
                    await session.flush()

                    folders_created.append({
                        "id": str(folder.id),
                        "name": tag,
                        "tag": tag,
                    })

                    # Create APIEndpoint for each operation
                    for op in operations:
                        endpoint = APIEndpoint(
                            project_id=project_id,
                            folder_id=folder.id,
                            display_name=f"{op.get('method', 'GET')} {op.get('path', '/')}",
                            path=op.get("path", "/"),
                            method=op.get("method", "GET"),
                            summary=op.get("summary"),
                            description=op.get("description"),
                            parameters=op.get("parameters", []),
                            request_body=op.get("request_body"),
                            responses=op.get("responses", {}),
                            tags=[tag],
                            tag_group=tag,
                        )
                        session.add(endpoint)
                        endpoints_created += 1

                await session.commit()

            return {
                "title": parsed.get("title", ""),
                "version": parsed.get("version", ""),
                "base_url": parsed.get("base_url", ""),
                "tags_created": len(folders_created),
                "endpoints_created": endpoints_created,
                "folders": folders_created,
            }
    ```

    ```python
    class APITestExecutor:
        """Async Playwright test execution engine.

        Pipeline:
        1. Get script from storage (workspace/api/scripts/)
        2. Write to temp directory
        3. Generate playwright.config.ts
        4. Run npx playwright test --reporter=json
        5. Parse JSON results
        6. Update APITestRun with stats
        7. Save individual APITestResult records
        """

        async def execute(self, run_id: UUID) -> None:
            """Execute a test run. Called as background task."""
            svc = APITestService()
            try:
                await svc.update_run_status(run_id, "running")
                run = await svc.get_test_run(run_id)
                test = await svc.get_api_test(run.api_test_id)

                # 1. Get script content
                script_content = await svc.get_script(run.api_test_id)
                if not script_content:
                    raise ValueError("No script found for test")

                # 2. Create temp execution directory
                exec_dir = Path(tempfile.mkdtemp(prefix="api_test_"))

                # 3. Write script file
                script_path = exec_dir / f"{test.identifier}.spec.ts"
                script_path.write_text(script_content, encoding="utf-8")

                # 4. Generate playwright.config.ts (minimal)
                config_content = self._generate_config(exec_dir)
                (exec_dir / "playwright.config.ts").write_text(config_content)

                # 5. Run npx playwright test
                start_time = time.time()
                proc = await asyncio.create_subprocess_exec(
                    "npx", "playwright", "test",
                    "--reporter=json",
                    f"--output={exec_dir / 'results'}",
                    str(script_path),
                    cwd=str(exec_dir),
                    stdout=asyncio.subprocess.PIPE,
                    stderr=asyncio.subprocess.PIPE,
                )
                stdout, stderr = await proc.communicate()
                duration_ms = int((time.time() - start_time) * 1000)

                # 6. Parse JSON results
                results = self._parse_results(stdout.decode())

                # 7. Update run and save results
                await svc.update_run_status(
                    run_id,
                    "completed" if proc.returncode == 0 else "failed",
                    total_tests=results.get("total", 0),
                    passed_tests=results.get("passed", 0),
                    failed_tests=results.get("failed", 0),
                    skipped_tests=results.get("skipped", 0),
                    duration_ms=duration_ms,
                )
                await svc.save_test_results(run_id, results.get("suites", []))

            except Exception as e:
                await svc.update_run_status(run_id, "failed", error_message=str(e))
            finally:
                # Cleanup temp dir
                shutil.rmtree(exec_dir, ignore_errors=True)

        def _generate_config(self, output_dir: Path) -> str:
            """Generate minimal playwright.config.ts."""
            return f"""
            import {{ defineConfig }} from '@playwright/test';
            export default defineConfig({{
              testDir: '.',
              reporter: [['json', {{ outputFile: '{output_dir / "results" / "results.json' }} }}]],
              use: {{ baseURL: 'http://localhost:3000' }},
            }});
            """

        def _parse_results(self, raw_output: str) -> dict:
            """Parse Playwright JSON reporter output."""
            try:
                data = json.loads(raw_output)
                suites = []
                for suite in data.get("suites", []):
                    for spec in suite.get("specs", []):
                        for test in spec.get("tests", []):
                            suites.append({
                                "scenario_name": spec.get("title", ""),
                                "status": test.get("status", "skipped"),
                                "duration_ms": test.get("duration", 0),
                                "error_message": test.get("results", [{}])[-1].get("error", {}).get("message"),
                            })
                passed = sum(1 for s in suites if s["status"] == "passed")
                failed = sum(1 for s in suites if s["status"] == "failed")
                return {
                    "total": len(suites),
                    "passed": passed,
                    "failed": failed,
                    "skipped": len(suites) - passed - failed,
                    "suites": suites,
                }
            except json.JSONDecodeError:
                return {"total": 0, "passed": 0, "failed": 0, "skipped": 0, "suites": []}
    ```

    ```python
    class ScenarioExecutionEngine:
        """Multi-step scenario execution engine.

        For each step in a scenario:
        1. Resolve data dependencies (JSONPath extraction from prior steps)
        2. Build HTTP request (method, url, headers, body)
        3. Send request via httpx.AsyncClient
        4. Extract data from response using JSONPath
        5. Run assertions (eq, ne, gt, lt, contains)
        6. Record result
        """

        def __init__(self):
            self.http_client = httpx.AsyncClient(timeout=30.0)

        async def execute_scenario(self, scenario_run_id: UUID) -> None:
            """Execute a scenario run. Called as background task."""
            svc = ScenarioService()
            try:
                run = await svc.get_scenario_run(scenario_run_id)
                scenario = await svc.get_scenario(run.scenario_id)

                context = ExecutionContext()  # variables + step_data
                results = []
                start_time = time.time()

                for step in scenario.steps:
                    step_result = await self._execute_step(
                        step, context, scenario, svc
                    )
                    results.append(step_result)

                    if step_result["status"] == "failed" and not step.continue_on_failure:
                        # Mark remaining steps as skipped
                        for remaining in scenario.steps[step.step_order:]:
                            results.append({"step_order": remaining.step_order, "status": "skipped"})
                        break

                duration_ms = int((time.time() - start_time) * 1000)
                passed = sum(1 for r in results if r["status"] == "passed")
                failed = sum(1 for r in results if r["status"] == "failed")
                skipped = sum(1 for r in results if r["status"] == "skipped")

                await svc.save_step_results(scenario_run_id, results)
                # Update run stats (use session directly)

            except Exception as e:
                # Record failure
                pass

        async def _execute_step(self, step, context, scenario, svc) -> dict:
            """Execute a single scenario step."""
            # 1. Resolve data dependencies
            resolver = DataDependencyResolver(context)
            for mapping in step.data_mappings:
                await resolver.resolve(mapping)

            # 2. Build request from step.request_override + resolved data
            request_config = resolver.apply_to_request(step.request_override)
            url = request_config.get("url", "")
            method = request_config.get("method", "GET")
            headers = request_config.get("headers", {})
            body = request_config.get("body")

            # 3. Apply delay if configured
            if step.delay_ms > 0:
                await asyncio.sleep(step.delay_ms / 1000)

            # 4. Send HTTP request with retry
            last_error = None
            for attempt in range(step.retry_count + 1):
                try:
                    response = await self.http_client.request(
                        method=method,
                        url=url,
                        headers=headers,
                        json=body,
                    )
                    break
                except Exception as e:
                    last_error = e
                    if attempt < step.retry_count:
                        await asyncio.sleep(1)

            if last_error and not response:
                return {
                    "step_order": step.step_order,
                    "status": "failed",
                    "error_message": str(last_error),
                }

            # 5. Extract data from response
            response_data = {
                "status_code": response.status_code,
                "headers": dict(response.headers),
                "body": response.json() if "json" in response.headers.get("content-type", "") else response.text,
            }
            context.set_step_data(str(step.id), response_data)

            # 6. Run assertions
            assertion_results = []
            for assertion in step.assertions:
                result = self._evaluate_assertion(assertion, response_data, context)
                assertion_results.append(result)

            all_passed = all(r.get("passed", False) for r in assertion_results)
            return {
                "step_order": step.step_order,
                "status": "passed" if all_passed else "failed",
                "request_data": request_config,
                "response_data": response_data,
                "assertion_results": assertion_results,
                "duration_ms": ...,
            }

        def _evaluate_assertion(self, assertion: dict, response: dict, context) -> dict:
            """Evaluate a single assertion against response data."""
            source = assertion.get("source", "response")
            path = assertion.get("path", "")
            operator = assertion.get("operator", "eq")
            expected = assertion.get("expected")

            # Extract actual value using JSONPath
            actual = self._extract_jsonpath(response, path) if path else response

            # Compare
            passed = False
            if operator == "eq": passed = actual == expected
            elif operator == "ne": passed = actual != expected
            elif operator == "gt": passed = actual > expected
            elif operator == "lt": passed = actual < expected
            elif operator == "contains": passed = expected in str(actual)
            elif operator == "matches": passed = bool(re.match(expected, str(actual)))

            return {"passed": passed, "actual": actual, "expected": expected, "operator": operator}

        def _extract_jsonpath(self, data: dict, path: str):
            """Simple JSONPath extraction (dot notation only)."""
            keys = path.split(".")
            current = data
            for key in keys:
                if isinstance(current, dict):
                    current = current.get(key)
                elif isinstance(current, list) and key.isdigit():
                    current = current[int(key)]
                else:
                    return None
            return current
    ```
